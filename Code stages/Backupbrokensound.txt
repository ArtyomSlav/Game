audio_ready = False
active_sounds = []
menu_music_playing = False

if pygame is not None:
    try:
        pygame.mixer.init()
        audio_ready = True
    except Exception as error:
        print("Audio unavailable:", error)


def play_sound(filename):
    if not audio_ready:
        return

    path = ASSETS / filename

    if not path.exists():
        print("Missing sound:", path)
        return

    try:
        sound = pygame.mixer.Sound(str(path))
        sound.play()
        active_sounds.append(sound)
    except Exception as error:
        print("Sound error:", error)


def stop_sounds():
    if audio_ready:
        pygame.mixer.stop()
    active_sounds.clear()


def play_menu_music():
    global menu_music_playing

    if not audio_ready or menu_music_playing:
        return

    path = ASSETS / "MenuMusic.mp3"

    if not path.is_file():
        print("Missing menu music:", path)
        return

    try:
        pygame.mixer.music.load(str(path))
        pygame.mixer.music.set_volume(0.5)
        pygame.mixer.music.play(-1)
        menu_music_playing = True
    except Exception as error:
        print("Menu music error:", error)


def stop_menu_music():
    global menu_music_playing

    if audio_ready:
        pygame.mixer.music.stop()

    menu_music_playing = False


SCENES = {
    "menu": "Menu.png",
    "Roadside": "Roadside.png",
    "mother_house": "mother_house.png",
    "endcredits": "endcredits.png"
}

image_cache = {}
background_photo = None


def get_image(name):
    if name == "black":
        return None

    if name in image_cache:
        return image_cache[name]

    filename = SCENES.get(name)

    if filename is None:
        return None

    path = ASSETS / filename

    if not path.exists():
        print("Missing image:", path)
        image_cache[name] = None
        return None

    try:
        image = Image.open(path).convert("RGB")
        image_cache[name] = image
        return image
    except Exception as error:
        print("Image error:", error)
        return None


def draw_background(name):
    global background_photo

    canvas.configure(bg="#000000")
    image = get_image(name)

    if image is None:
        return

    w = max(canvas.winfo_width(), 1)
    h = max(canvas.winfo_height(), 1)

    resized = ImageOps.fit(
        image,
        (w, h),
        method=Image.Resampling.LANCZOS
    )

    background_photo = ImageTk.PhotoImage(resized)

    canvas.create_image(
        0,
        0,
        image=background_photo,
        anchor="nw"
    )

def slot_path(slot):
    if not isinstance(slot, int) or not 1 <= slot <= SAVE_SLOTS:
        raise ValueError("Invalid save slot")
    return SAVE_DIR / f"slot_{slot}.json"


def capture_progress():
    if mode not in ("story", "choice"):
        return None

    return {
        "version": SAVE_VERSION,
        "label": current_label,
        "index": story_index,
        "scene": current_scene,
        "mode": mode,
        "line": list(current_line) if current_line else None,
        "choices": [list(choice) for choice in current_choices],
        "saved_at": datetime.now().isoformat(timespec="seconds")
    }


def read_slot(slot):
    path = slot_path(slot)

    if not path.exists():
        return None

    try:
        with path.open("r", encoding="utf-8") as file:
            data = json.load(file)

        if not isinstance(data, dict):
            raise ValueError("Invalid save data")

        if data.get("version") != SAVE_VERSION:
            raise ValueError("Unsupported save version")

        label = data["label"]
        index = data["index"]
        saved_mode = data["mode"]
        scene = data["scene"]

        if label not in STORY:
            raise ValueError("Unknown story route")

        if type(index) is not int:
            raise ValueError("Invalid story index")

        if not 0 <= index <= len(STORY[label]):
            raise ValueError("Story index out of range")

        if saved_mode not in ("story", "choice"):
            raise ValueError("Invalid game mode")

        if scene not in SCENES and scene != "black":
            raise ValueError("Invalid scene")

        if saved_mode == "story":
            line = data.get("line")

            if (
                not isinstance(line, list)
                or len(line) != 2
                or line[0] not in CHARACTERS
                or not isinstance(line[1], str)
            ):
                raise ValueError("Invalid dialogue")

            if index < 1:
                raise ValueError("Invalid dialogue position")

            expected = STORY[label][index - 1]

            if (
                len(expected) != 2
                or expected[0] != line[0]
                or expected[1] != line[1]
            ):
                raise ValueError("Dialogue does not match story")

        if saved_mode == "choice":
            choices = data.get("choices")

            if not isinstance(choices, list) or not choices:
                raise ValueError("Invalid choices")

            for choice in choices:
                if (
                    not isinstance(choice, list)
                    or len(choice) != 2
                    or not isinstance(choice[0], str)
                    or not isinstance(choice[1], str)
                    or choice[1] not in STORY
                ):
                    raise ValueError("Invalid choice")

            if index < 1:
                raise ValueError("Invalid choice position")

            expected = STORY[label][index - 1]

            if (
                expected[0] != "choice"
                or expected[1] != [tuple(c) for c in choices]
            ):
                raise ValueError("Choices do not match story")

        return data

    except (
        OSError, ValueError, KeyError,
        TypeError, IndexError
    ) as error:
        print("Save file error:", error)
        return None


def write_slot(slot):
    if save_snapshot is None:
        return

    data = dict(save_snapshot)
    data["saved_at"] = datetime.now().isoformat(timespec="seconds")

    path = slot_path(slot)
    temporary = path.with_suffix(".tmp")

    try:
        SAVE_DIR.mkdir(parents=True, exist_ok=True)

        with temporary.open("w", encoding="utf-8") as file:
            json.dump(
                data,
                file,
                indent=4,
                ensure_ascii=False
            )

        os.replace(temporary, path)

        messagebox.showinfo(
            "Game Saved",
            f"Progress saved successfully to Slot {slot}."
        )

        return_from_slots()

    except OSError as error:
        messagebox.showerror("Save Error", str(error))


def restore_slot(slot):
    global mode, current_label, story_index
    global current_scene, current_line, current_choices
    global save_snapshot

    data = read_slot(slot)

    if data is None:
        messagebox.showwarning(
            "LOAD",
            "This slot is empty or its save file is invalid."
        )
        return

    stop_menu_music()
    stop_sounds()

    current_label = data["label"]
    story_index = data["index"]
    current_scene = data["scene"]

    current_line = (
        tuple(data["line"]) if data.get("line") else None
    )

    current_choices = [
        tuple(choice) for choice in data.get("choices", [])
    ]

    mode = data["mode"]
    save_snapshot = None

    if mode == "choice":
        draw_choices()
    else:
        draw_story()


def open_slots(screen_type):
    global mode, previous_mode, save_snapshot

    if screen_type == "save":
        if mode not in ("story", "choice"):
            return

        save_snapshot = capture_progress()
        previous_mode = mode
        mode = "save"

    elif screen_type == "load":
        if mode not in ("menu", "story", "choice"):
            return

        previous_mode = mode
        save_snapshot = None
        mode = "load"

    else:
        return

    draw_slots()


def draw_slots():
    clear_canvas()
    canvas.configure(cursor="", bg=BG)

    w = canvas.winfo_width()
    h = canvas.winfo_height()

    title = "SAVE GAME" if mode == "save" else "LOAD GAME"

    canvas.create_text(
        w / 2,
        h * 0.13,
        text=title,
        fill=BLUE,
        font=(FONT, 26, "bold")
    )

    for slot in range(1, SAVE_SLOTS + 1):
        data = read_slot(slot)
        y = h * 0.27 + (slot - 1) * h * 0.19
        tag = f"slot_{slot}"

        canvas.create_rectangle(
            w * 0.20,
            y - 42,
            w * 0.80,
            y + 42,
            fill="#141A22",
            outline="#607F93",
            tags=(tag,)
        )

        if data:
            route = data["label"].upper()

            if data.get("line"):
                preview = data["line"][1]
            else:
                preview = "Story choice"

            preview = preview.replace("{b}", "").replace("{/b}", "")
            preview = preview[:65]

            try:
                date = datetime.fromisoformat(
                    data["saved_at"]
                ).strftime("%b %d, %Y  %I:%M %p")
            except (ValueError, KeyError, TypeError):
                date = "Unknown date"

            description = f"{route}  |  {date}\n{preview}"

        else:
            description = "EMPTY SLOT"

        canvas.create_text(
            w * 0.24,
            y,
            text=f"SLOT {slot:02d}",
            fill=BLUE,
            font=(FONT, 16, "bold"),
            anchor="w",
            tags=(tag,)
        )

        canvas.create_text(
            w * 0.40,
            y,
            text=description,
            fill=TEXT,
            font=(FONT, 12),
            anchor="w",
            width=w * 0.37,
            tags=(tag,)
        )

        canvas.tag_bind(
            tag,
            "<Enter>",
            lambda event, t=tag: slot_hover(t, True)
        )

        canvas.tag_bind(
            tag,
            "<Leave>",
            lambda event, t=tag: slot_hover(t, False)
        )

        canvas.tag_bind(
            tag,
            "<Button-1>",
            lambda event, s=slot: select_slot(s)
        )

    canvas.create_text(
        w / 2,
        h * 0.90,
        text="ESC — BACK",
        fill=MUTED,
        font=(FONT, 13)
    )


def slot_hover(tag, active):
    for item in canvas.find_withtag(tag):
        if canvas.type(item) == "rectangle":
            canvas.itemconfigure(
                item,
                outline=BLUE if active else "#607F93",
                fill=HOVER_BG if active else "#141A22"
            )

    canvas.configure(cursor="hand2" if active else "")


def select_slot(slot):
    if mode == "save":
        path = slot_path(slot)

        if path.exists():
            confirm = messagebox.askyesno(
                "Overwrite Save",
                f"Overwrite Slot {slot}?"
            )

            if not confirm:
                return

        write_slot(slot)

    elif mode == "load":
        restore_slot(slot)


def return_from_slots():
    global mode, save_snapshot

    mode = previous_mode
    save_snapshot = None

    if mode == "menu":
        draw_menu()
    elif mode == "story":
        draw_story()
    elif mode == "choice":
        draw_choices()

def start_game():
    global mode, current_label, story_index
    global current_scene, current_line, current_choices

    # Display the content warning before starting a new game.
    if not show_content_disclaimer():
        return

    stop_menu_music()
    stop_sounds()

    mode = "story"
    current_label = "start"
    story_index = 0
    current_scene = "black"
    current_line = None
    current_choices = []

    process_story()


def process_story():
    global mode, story_index, current_scene
    global current_line, current_choices

    if mode != "story":
        return

    script = STORY[current_label]

    while story_index < len(script):
        entry = script[story_index]
        story_index += 1

        command = entry[0]

        if command == "scene":
            current_scene = entry[1]
            continue

        if command == "sound":
            play_sound(entry[1])
            continue

        if command == "choice":
            current_choices = entry[1]
            current_line = None
            mode = "choice"
            draw_choices()
            return

        if command == "end":
            mode = "ending"
            draw_ending()
            return

        current_line = entry
        draw_story()
        return

    mode = "ending"
    draw_ending()


def draw_story():
    global dialogue_box

    clear_canvas()
    canvas.configure(cursor="")

    draw_background(current_scene)

    if current_line is None:
        return

    speaker_id, dialogue = current_line
    speaker = CHARACTERS.get(speaker_id, speaker_id or "")

    w = canvas.winfo_width()
    h = canvas.winfo_height()

    x1 = w * 0.055
    x2 = w * 0.945
    y1 = h * 0.69
    y2 = h * 0.94

    canvas.create_rectangle(
        x1, y1, x2, y2,
        fill="#090D14",
        outline="#607F93",
        width=1
    )

    if speaker:
        canvas.create_text(
            x1 + 30,
            y1 + 30,
            text=speaker,
            fill=BLUE,
            font=(FONT, 19, "bold"),
            anchor="w"
        )

    dialogue_box = tk.Text(
        canvas,
        bg="#090D14",
        fg=TEXT,
        font=(FONT, 16),
        wrap="word",
        bd=0,
        highlightthickness=0,
        padx=0,
        pady=0,
        cursor="hand2",
        takefocus=0
    )

    dialogue_box.tag_configure(
        "bold",
        font=(FONT, 16, "bold")
    )

    parts = re.split(r"(\{b\}|\{/b\})", dialogue)
    bold_active = False

    for part in parts:
        if part == "{b}":
            bold_active = True
        elif part == "{/b}":
            bold_active = False
        elif part:
            if bold_active:
                dialogue_box.insert("end", part, "bold")
            else:
                dialogue_box.insert("end", part)

    dialogue_box.configure(state="disabled")

    canvas.create_window(
        x1 + 30,
        y1 + 73,
        window=dialogue_box,
        anchor="nw",
        width=max(int(x2 - x1 - 60), 100),
        height=max(int(y2 - y1 - 95), 35)
    )

    dialogue_box.bind(
        "<Button-1>",
        lambda event: next_dialogue()
    )

    canvas.create_text(
        x2 - 25,
        y2 - 18,
        text="▼",
        fill=BLUE,
        font=(FONT, 12)
    )


def next_dialogue(event=None):
    if mode != "story":
        return

    process_story()


def draw_choices():

    canvas.configure(cursor="")

    draw_background(current_scene)

    w = canvas.winfo_width()
    h = canvas.winfo_height()
    x = w / 2

    canvas.create_text(
        x,
        h * 0.35,
        text="CHOOSE",
        fill=BLUE,
        font=(FONT, 16)
    )

    for i, (option, destination) in enumerate(current_choices):
        y = h * 0.46 + i * 85
        tag = f"choice_{i}"

        canvas.create_rectangle(
            x - 230,
            y - 27,
            x + 230,
            y + 27,
            fill=PANEL,
            outline="#607F93",
            tags=(tag,)
        )

        canvas.create_text(
            x,
            y,
            text=option,
            fill=TEXT,
            font=(FONT, 17),
            tags=(tag,)
        )

        canvas.tag_bind(
            tag,
            "<Enter>",
            lambda event, t=tag: choice_hover(t, True)
        )

        canvas.tag_bind(
            tag,
            "<Leave>",
            lambda event, t=tag: choice_hover(t, False)
        )

        canvas.tag_bind(
            tag,
            "<Button-1>",
            lambda event, d=destination: jump_to(d)
        )


def choice_hover(tag, active):
    for item in canvas.find_withtag(tag):
        if canvas.type(item) == "text":
            canvas.itemconfigure(
                item,
                fill=BLUE if active else TEXT
            )

    canvas.configure(cursor="hand2" if active else "")


def jump_to(destination):
    global current_label, story_index, mode
    global current_choices, current_line

    if destination not in STORY:
        print("Unknown story label:", destination)
        return

    current_label = destination
    story_index = 0
    current_choices = []
    current_line = None
    mode = "story"

    process_story()


def draw_ending():
    clear_canvas()
    canvas.configure(cursor="")

    draw_background(current_scene)

    w = canvas.winfo_width()
    h = canvas.winfo_height()

    if current_scene != "endcredits":
        canvas.create_text(
            w / 2,
            h / 2,
            text="END",
            fill=BLUE,
            font=(FONT, 28)
        )

    canvas.create_text(
        w / 2,
        h * 0.88,
        text="CLICK TO RETURN TO MENU",
        fill=TEXT,
        font=(FONT, 13)
    )


def handle_click(event):
    if mode == "story":
        next_dialogue()
    elif mode == "ending":
        draw_menu()


def handle_key(event):
    key = event.keysym.lower()

    if key == "escape":
        if mode in ("save", "load"):
            return_from_slots()
        elif mode != "menu":
            draw_menu()
        return

    if key == "s":
        if mode in ("story", "choice"):
            open_slots("save")
        return

    if key == "l":
        if mode in ("menu", "story", "choice"):
            open_slots("load")
        return

    if key in ("space", "return"):
        if mode == "story":
            next_dialogue()
        elif mode == "ending":
            draw_menu()


canvas.bind("<Button-1>", handle_click)
app.bind("<Key>", handle_key)


def redraw():
    global resize_job
    resize_job = None

    if mode == "menu":
        draw_menu()
    elif mode == "story":
        draw_story()
    elif mode == "choice":
        draw_choices()
    elif mode == "ending":
        draw_ending()
    elif mode in ("save", "load"):
        draw_slots()


def on_resize(event):
    global resize_job

    if event.widget != canvas:
        return

    if resize_job is not None:
        app.after_cancel(resize_job)

    resize_job = app.after(120, redraw)


canvas.bind("<Configure>", on_resize)


def close_app():
    stop_menu_music()
    stop_sounds()

    if audio_ready:
        pygame.mixer.quit()

    app.destroy()


app.protocol("WM_DELETE_WINDOW", close_app)

app.after(100, draw_menu)
app.mainloop()
